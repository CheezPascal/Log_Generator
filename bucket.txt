    <script>
        // Content from the files
        const fileAContent = [
            "I found some challenges revisiting Kubernetes networking concepts...",
            "I didn't find any significant challenges this week...",
            "I found it challenging to grasp some of the more advanced Terraform concepts...",
            // Add all 70 entries from file_A.txt here
        ];

        const fileBContent = [
            "I found it most enjoyable successfully deploying a multi-container application...",
            "I didn't encounter anything particularly enjoyable this week...",
            "I found it most enjoyable deepening my understanding of Terraform modules...",
            // Add all 70 entries from file_B.txt here
        ];

        const fileCContent = [
            "Working on my resume profile to highlight my recent Kubernetes and Docker skills...",
            "No other notable comments.",
            "Started working on a personal project to automate the deployment...",
            // Add all 70 entries from file_C.txt here
        ];

        // Function to randomize text from a file
        function getRandomText(fileContent) {
            return fileContent[Math.floor(Math.random() * fileContent.length)];
        }

        // Function to populate the boxes with random text
        function populateBoxes() {
            document.getElementById('box1').innerText = getRandomText(fileAContent);
            document.getElementById('box2').innerText = getRandomText(fileBContent);
            document.getElementById('box3').innerText = getRandomText(fileCContent);
        }

        // Function to copy text from all boxes to clipboard
        function copyToClipboard() {
            const box1Text = document.getElementById('box1').innerText;
            const box2Text = document.getElementById('box2').innerText;
            const box3Text = document.getElementById('box3').innerText;
            const combinedText = `Box 1: ${box1Text}\nBox 2: ${box2Text}\nBox 3: ${box3Text}`;
            navigator.clipboard.writeText(combinedText).then(() => {
                alert('Copied to clipboard!');
            }).catch(() => {
                alert('Failed to copy!');
            });
        }

        // Event listeners for buttons
        document.getElementById('randomizeBtn').addEventListener('click', populateBoxes);
        document.getElementById('copyBtn').addEventListener('click', copyToClipboard);

        // Initial population of boxes
        populateBoxes();
    </script>
